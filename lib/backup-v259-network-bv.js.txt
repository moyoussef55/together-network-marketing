import { db } from "hatchable";
import { requireNetworkUser } from "../../lib/network-auth.js";

export const access = "public";
export const methods = ["GET"];

export default async function handler(req,res){
  const user=await requireNetworkUser(req,res);
  if(!user)return;
  const uid=user.id;
  const requested=String(req.query?.member_id||uid).trim();
  const position=Number(req.query?.account_position||1);
  if(![1,2,3].includes(position)) return res.status(400).json({error:"invalid_account_position"});

  if(requested!==uid){
    const allowed=await db.query(`WITH RECURSIVE descendants AS (
      SELECT p.member_id::text AS member_id FROM network_placements p WHERE p.owner_marketer_id=$1
      UNION SELECT p2.member_id::text FROM network_placements p2 JOIN descendants d ON p2.owner_marketer_id::text=d.member_id
    ), ancestors AS (
      SELECT p.owner_marketer_id::text AS member_id FROM network_placements p WHERE p.member_id::text=$1
      UNION SELECT p2.owner_marketer_id::text FROM network_placements p2 JOIN ancestors a ON p2.member_id::text=a.member_id
    ) SELECT 1 WHERE $2::text IN (SELECT member_id FROM descendants) OR $2::text IN (SELECT member_id FROM ancestors) LIMIT 1`,[uid,requested]);
    if(!allowed.rows.length) return res.status(403).json({error:"network_access_denied"});
  }
  const balance=await db.query(`SELECT member_id,account_position,total_right,total_left,carry_right,carry_left,updated_at FROM network_bv_balances WHERE member_id=$1 AND account_position=$2`,[requested,position]);
  const b=balance.rows[0]||{member_id:requested,account_position:position,total_right:0,total_left:0,carry_right:0,carry_left:0};
  const carryView=await db.query(`WITH last_settlement AS (SELECT MAX(week_end)::date AS week_end FROM network_bv_settlements WHERE member_id=$1 AND account_position=$2), cycle_start AS (SELECT COALESCE((SELECT week_end FROM last_settlement),(date_trunc('week',timezone('Africa/Cairo',now()))::date - 2)) AS start_date), period_bv AS (SELECT COALESCE(SUM(bv) FILTER(WHERE side='right'),0)::bigint AS right_bv,COALESCE(SUM(bv) FILTER(WHERE side='left'),0)::bigint AS left_bv FROM network_bv_ledger,cycle_start WHERE upline_member_id=$1 AND account_position=$2 AND sale_at>=cycle_start.start_date), combined AS (SELECT (COALESCE($3::bigint,0)+right_bv) AS right_total,(COALESCE($4::bigint,0)+left_bv) AS left_total FROM period_bv) SELECT right_total AS carry_right,left_total AS carry_left FROM combined`,[requested,position,b.carry_right||0,b.carry_left||0]);
  if(carryView.rows[0]){b.carry_right=Number(carryView.rows[0].carry_right||0);b.carry_left=Number(carryView.rows[0].carry_left||0);}
  const checks=await db.query(`SELECT id,week_start,week_end,right_bv,left_bv,matched_bv,check_amount,carry_right,carry_left,settled_at FROM network_bv_settlements WHERE member_id=$1 AND account_position=$2 ORDER BY settled_at DESC LIMIT 12`,[requested,position]);
  const recent=await db.query(`SELECT order_id,bv,side,sale_at,source_member_id FROM network_bv_ledger WHERE upline_member_id=$1 AND account_position=$2 ORDER BY sale_at DESC LIMIT 100`,[requested,position]);
  return res.json({ok:true,balance:b,checks:checks.rows,recent_sales:recent.rows});
}
